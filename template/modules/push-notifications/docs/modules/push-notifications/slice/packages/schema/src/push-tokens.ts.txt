// push-notifications module — the device push-token table + RLS declarations.
// Installed to packages/schema/src/push-tokens.ts and re-exported from
// index.ts so drizzle-kit (schema: './src/index.ts') and the schema-rls gate
// both see it (see the module's APPLY.md). The wire DTOs live in @app/contracts
// (appended to its index); the SQL side is drizzle/0003_push_device_tokens.sql.
import { sql } from 'drizzle-orm'
import type { AnyPgColumn } from 'drizzle-orm/pg-core'
// SOURCE: drizzle pgPolicy/pgRole declare RLS in the schema so the schema-rls
// gate can assert every table carries policies [corpus: postgres/rls-force]
import { pgPolicy, pgRole, pgTable, text, timestamp, uuid } from 'drizzle-orm/pg-core'

// Runtime login role, created by the docker-compose init SQL — hence
// `.existing()` (same declaration shape as packages/schema/src/index.ts; the
// role objects are name-keyed, so redeclaring it here creates nothing).
const appApi = pgRole('app_api').existing()

// The initPlan-shaped owner predicate, identical to the notes table's:
// scalar sub-select so identity resolves once per statement, nullif('') so
// both no-identity shapes fail closed instead of raising 22P02.
// SOURCE: PostgreSQL row-security guidance [corpus: postgres/rls-initplan]
const ownerIsCurrentUser = (ownerId: AnyPgColumn) =>
  sql`${ownerId} = (select nullif(current_setting('app.user_id', true), '')::uuid)`

/**
 * One row per (owner, device push token). The id is DETERMINISTIC — a
 * version-5 UUID of (owner_id, token) computed by the DAL
 * (apps/server/src/dal/push-token-id.ts) — which is what makes register() an
 * idempotent, race-free single-statement upsert with no second unique index.
 * The DEFAULT stays gen_random_uuid() so harness probes (which bulk-seed
 * without ids) and ad-hoc inserts still work.
 *
 * `platform` is constrained by a CHECK in the migration and by the Zod enum
 * in @app/contracts (PUSH_PLATFORMS) — deliberately NOT a drizzle-level enum:
 * the DB and the wire are the enforcement surfaces, and the DAL's exit parse
 * rejects anything the CHECK somehow let through.
 */
export const pushDeviceTokens = pgTable(
  'push_device_tokens',
  {
    createdAt: timestamp('created_at', { mode: 'string', withTimezone: true })
      .notNull()
      .defaultNow(),
    id: uuid('id').primaryKey().defaultRandom(),
    ownerId: uuid('owner_id').notNull(),
    platform: text('platform').notNull(),
    token: text('token').notNull(),
    updatedAt: timestamp('updated_at', { mode: 'string', withTimezone: true })
      .notNull()
      .defaultNow(),
  },
  (table) => [
    // Four per-operation policies (never FOR ALL) — same doctrine, and the same
    // per-op citations, as the notes table.
    // SOURCE: per-command policies, USING filters reads [corpus: postgres/rls-force]; per-op doctrine [corpus: harness/doctrine]
    pgPolicy('push_device_tokens_select_own', {
      as: 'permissive',
      for: 'select',
      to: appApi,
      using: ownerIsCurrentUser(table.ownerId),
    }),
    // SOURCE: per-op owner policy — insert guards via WITH CHECK; ON CONFLICT DO UPDATE
    // additionally checks the UPDATE policies on the conflicting row
    // https://www.postgresql.org/docs/current/sql-createpolicy.html [corpus: postgres/rls-force]
    pgPolicy('push_device_tokens_insert_own', {
      as: 'permissive',
      for: 'insert',
      to: appApi,
      withCheck: ownerIsCurrentUser(table.ownerId),
    }),
    // SOURCE: per-op owner policy — update guards read AND write rows [corpus: postgres/rls-force]
    pgPolicy('push_device_tokens_update_own', {
      as: 'permissive',
      for: 'update',
      to: appApi,
      using: ownerIsCurrentUser(table.ownerId),
      withCheck: ownerIsCurrentUser(table.ownerId),
    }),
    // SOURCE: per-op owner policy — delete scoped by USING [corpus: postgres/rls-force]
    pgPolicy('push_device_tokens_delete_own', {
      as: 'permissive',
      for: 'delete',
      to: appApi,
      using: ownerIsCurrentUser(table.ownerId),
    }),
  ],
).enableRLS()
