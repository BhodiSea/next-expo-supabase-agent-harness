import {
  ApiError,
  type NewPushDeviceToken,
  NewPushDeviceTokenInput,
  PushDeviceTokenDto,
  PushDeviceTokensListQuery,
  PushDeviceTokensPage,
} from '@app/contracts'
import { createRoute, type OpenAPIHono, z } from '@hono/zod-openapi'
import { decodeNotesCursor } from '../dal/cursor.js'
import { apiError } from '../errors.js'
import type { AppEnv, PushTokensDal } from '../types.js'

// push-notifications module — the /api/push/tokens route slice. Installed to
// apps/server/src/routes/push-tokens.ts and registered from app.ts AFTER the
// /api/* middleware mounts (see APPLY.md), so every route here sits behind the
// same skew + auth + body-limit guards as the notes routes — the middleware
// walk in middleware/skew.test.ts proves it, and `pnpm --filter server
// openapi:emit` must be re-run after registering (the contracts gate diffs the
// committed document).

// Stryker disable all — OpenAPI route DECLARATIONS, not behaviour (same control
// as app.ts's route table): drift is caught byte-for-byte by the `contracts`
// gate and the spec walk in app.errors.test.ts. The handlers stay OUTSIDE.
const errorResponse = (description: string) => ({
  content: { 'application/json': { schema: ApiError } },
  description,
})

// The same guarded-route failure set every authenticated /api route declares.
const guardedRouteErrors = {
  400: errorResponse('Request validation failed (envelope code bad_request)'),
  401: errorResponse('Missing or unverifiable bearer token'),
  409: errorResponse('Client major version does not match the server (code version_skew)'),
  500: errorResponse('Unexpected server error — correlate via error.requestId'),
}

// z.guid() matches the postgres uuid type (any 8-4-4-4-12 hex, no variant check).
const PushTokenParamsDto = z.object({ id: z.guid() })

const registerPushTokenRoute = createRoute({
  method: 'post',
  path: '/api/push/tokens',
  security: [{ Bearer: [] }],
  request: {
    body: {
      required: true,
      content: { 'application/json': { schema: NewPushDeviceTokenInput } },
    },
  },
  responses: {
    201: {
      description:
        'The registered device token row. Idempotent per (owner, token): re-registering ' +
        'the same token answers 201 with the SAME row id — clients keep that id to ' +
        'deregister on sign-out',
      content: { 'application/json': { schema: PushDeviceTokenDto } },
    },
    ...guardedRouteErrors,
    413: errorResponse('Request body exceeds the 1 MiB /api/* limit'),
  },
})

const listPushTokensRoute = createRoute({
  method: 'get',
  path: '/api/push/tokens',
  security: [{ Bearer: [] }],
  request: { query: PushDeviceTokensListQuery },
  responses: {
    200: {
      description:
        'One keyset page of the device tokens owned by the authenticated user (scoped by ' +
        'RLS), newest first; follow nextCursor until it is null',
      content: { 'application/json': { schema: PushDeviceTokensPage } },
    },
    ...guardedRouteErrors,
  },
})

const removePushTokenRoute = createRoute({
  method: 'delete',
  path: '/api/push/tokens/{id}',
  security: [{ Bearer: [] }],
  request: { params: PushTokenParamsDto },
  responses: {
    204: { description: 'Device token deregistered' },
    ...guardedRouteErrors,
    404: errorResponse('No such device token visible to this user'),
  },
})
// Stryker restore all

/** Registers the push-token routes; app.ts passes the real DAL (tests inject fakes). */
export function registerPushTokenRoutes(app: OpenAPIHono<AppEnv>, dal: PushTokensDal): void {
  app.openapi(registerPushTokenRoute, async (c) => {
    const input: NewPushDeviceToken = c.req.valid('json')
    const row = await dal.register(c.get('userId'), input)
    return c.json(row, 201)
  })

  app.openapi(listPushTokensRoute, async (c) => {
    const query = c.req.valid('query')
    const cursor = query.cursor === undefined ? undefined : decodeNotesCursor(query.cursor)
    if (cursor === null) {
      // Well-formed base64url that this server never minted — reject at the
      // edge, before the DAL sees it.
      return apiError(c, 400, 'bad_request', 'cursor is not a page token this server issued')
    }
    const page = await dal.list(c.get('userId'), { cursor, limit: query.limit })
    return c.json(page, 200)
  })

  app.openapi(removePushTokenRoute, async (c) => {
    const { id } = c.req.valid('param')
    const removed = await dal.remove(c.get('userId'), id)
    return removed ? c.body(null, 204) : apiError(c, 404, 'not_found', 'no such device token')
  })
}
