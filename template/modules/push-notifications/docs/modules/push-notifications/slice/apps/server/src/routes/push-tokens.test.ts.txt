// push-notifications module — route-wiring tests for /api/push/tokens, the
// app.routes.test.ts pattern: a recording fake DAL injected through
// createApp({ pushTokensDal }) pins the wiring in BOTH directions — down (the
// verified identity, the parsed input, the bounded limit, the DECODED cursor)
// and back (201 body, 400 on a foreign cursor, 204 vs 404 on delete).
import {
  type NewPushDeviceToken,
  PUSH_TOKENS_PAGE_LIMIT_DEFAULT,
  type PushDeviceToken,
  type PushDeviceTokensPageDto,
} from '@app/contracts'
import { describe, expect, it } from 'vitest'
import { type AppOptions, createApp } from '../app.js'
import { encodeNotesCursor, type NoteCursorKey } from '../dal/cursor.js'
import type { PushTokensDal } from '../types.js'

const USER_ID = '11111111-1111-4111-8111-111111111111'
const ROW_ID = '37129106-1b10-5d96-9bf7-237d0a74acfc'
const authed = { authorization: 'Bearer test-token' }

const stored: PushDeviceToken = {
  createdAt: '2026-01-01 00:00:00.123456+00',
  id: ROW_ID,
  ownerId: USER_ID,
  platform: 'ios',
  token: 'ExponentPushToken[demo-device-1]',
  updatedAt: '2026-01-01 00:00:00.123456+00',
}

interface DalCalls {
  readonly register: { userId: string; input: NewPushDeviceToken }[]
  readonly list: { userId: string; limit: number; cursor: NoteCursorKey | undefined }[]
  readonly remove: { userId: string; id: string }[]
}

interface DalResults {
  readonly registered?: PushDeviceToken
  readonly page?: PushDeviceTokensPageDto
  readonly removed?: boolean
}

/** A DAL that records every argument the routes hand it and replays canned results. */
function fixture(results: DalResults = {}): { options: AppOptions; calls: DalCalls } {
  const calls: DalCalls = { list: [], register: [], remove: [] }
  const pushTokensDal: PushTokensDal = {
    register: (userId, input) => {
      calls.register.push({ userId, input })
      return results.registered === undefined
        ? Promise.reject(new Error('register was not stubbed for this test'))
        : Promise.resolve(results.registered)
    },
    list: (userId, page) => {
      calls.list.push({ userId, limit: page.limit, cursor: page.cursor })
      return Promise.resolve(results.page ?? { items: [], nextCursor: null })
    },
    remove: (userId, id) => {
      calls.remove.push({ userId, id })
      return Promise.resolve(results.removed ?? false)
    },
  }
  return {
    options: {
      version: '1.2.3',
      verifyToken: () => Promise.resolve({ userId: USER_ID }),
      pushTokensDal,
    },
    calls,
  }
}

describe('POST /api/push/tokens', () => {
  it('hands the DAL the VERIFIED identity and the parsed input, answers 201 + the row', async () => {
    const { options, calls } = fixture({ registered: stored })

    const res = await createApp(options).request('/api/push/tokens', {
      method: 'POST',
      headers: { ...authed, 'content-type': 'application/json' },
      body: JSON.stringify({ platform: 'ios', token: stored.token }),
    })

    expect(res.status).toBe(201)
    expect(await res.json()).toEqual(stored)
    // Identity is the token subject — the wire carries no owner field at all.
    expect(calls.register).toEqual([
      { userId: USER_ID, input: { platform: 'ios', token: stored.token } },
    ])
  })

  it('rejects an unknown platform with the 400 envelope before the DAL is touched', async () => {
    const { options, calls } = fixture()

    const res = await createApp(options).request('/api/push/tokens', {
      method: 'POST',
      headers: { ...authed, 'content-type': 'application/json' },
      body: JSON.stringify({ platform: 'web', token: 'ExponentPushToken[demo-device-1]' }),
    })

    expect(res.status).toBe(400)
    const body = (await res.json()) as { error: { code: string } }
    expect(body.error.code).toBe('bad_request')
    expect(calls.register).toHaveLength(0)
  })

  it('401s without a bearer token — the guard covers this route like every /api route', async () => {
    const { options, calls } = fixture()

    const res = await createApp(options).request('/api/push/tokens', {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ platform: 'ios', token: 'ExponentPushToken[demo-device-1]' }),
    })

    expect(res.status).toBe(401)
    expect(calls.register).toHaveLength(0)
  })
})

describe('GET /api/push/tokens', () => {
  it('applies the contract default limit (never an unbounded scan)', async () => {
    const { options, calls } = fixture()

    const res = await createApp(options).request('/api/push/tokens', { headers: authed })

    expect(res.status).toBe(200)
    expect(calls.list).toEqual([
      { userId: USER_ID, limit: PUSH_TOKENS_PAGE_LIMIT_DEFAULT, cursor: undefined },
    ])
  })

  it('DECODES the opaque cursor before the DAL sees it, and returns the page verbatim', async () => {
    const key: NoteCursorKey = { createdAt: '2026-01-01 00:00:00.123456+00', id: ROW_ID }
    const nextCursor = encodeNotesCursor(key)
    const { options, calls } = fixture({ page: { items: [stored], nextCursor } })

    const res = await createApp(options).request(
      `/api/push/tokens?cursor=${encodeNotesCursor(key)}&limit=2`,
      { headers: authed },
    )

    expect(res.status).toBe(200)
    expect(await res.json()).toEqual({ items: [stored], nextCursor })
    expect(calls.list).toEqual([{ userId: USER_ID, limit: 2, cursor: key }])
  })

  it('400s a cursor this server never minted, before the DAL sees it', async () => {
    const { options, calls } = fixture()

    // Valid base64url alphabet (passes the Zod regex) but not our JSON keyset.
    const res = await createApp(options).request('/api/push/tokens?cursor=bm90LW91cnM', {
      headers: authed,
    })

    expect(res.status).toBe(400)
    const body = (await res.json()) as { error: { code: string; message: string } }
    expect(body.error.code).toBe('bad_request')
    expect(calls.list).toHaveLength(0)
  })
})

describe('DELETE /api/push/tokens/{id}', () => {
  it('204s when the DAL removed the row', async () => {
    const { options, calls } = fixture({ removed: true })

    const res = await createApp(options).request(`/api/push/tokens/${ROW_ID}`, {
      method: 'DELETE',
      headers: authed,
    })

    expect(res.status).toBe(204)
    expect(calls.remove).toEqual([{ userId: USER_ID, id: ROW_ID }])
  })

  it('404s with the envelope when nothing visible matched (foreign or absent id)', async () => {
    const { options } = fixture({ removed: false })

    const res = await createApp(options).request(`/api/push/tokens/${ROW_ID}`, {
      method: 'DELETE',
      headers: authed,
    })

    expect(res.status).toBe(404)
    const body = (await res.json()) as { error: { code: string } }
    expect(body.error.code).toBe('not_found')
  })
})
