// push-notifications module — N+1 regression gate for the push-token DAL, the
// dal/notes.statements.test.ts pattern applied to this slice: every method
// must execute a FIXED number of SQL statements no matter how many rows come
// back. The counter is the drizzle pg-proxy execution callback — every
// statement the DAL emits must pass through it to execute.
//
// Expected counts:
//   register() = exactly 1 — INSERT ... ON CONFLICT DO UPDATE ... RETURNING is
//                the whole upsert; no read-before-write, no follow-up SELECT.
//   list()     = exactly 1 — the single keyset SELECT.
//   remove()   = exactly 1 — DELETE ... RETURNING.
// The BEGIN / set_config / COMMIT wrapper lives in the real withUserContext
// (mocked here) and is constant per call; the rls lane proves it live.
import { pushDeviceTokens } from '@app/schema'
import { getTableColumns } from 'drizzle-orm'
import { drizzle } from 'drizzle-orm/pg-proxy'
import { describe, expect, it, vi } from 'vitest'
import type { UserTx } from '../db/context.js'

const state = vi.hoisted(() => ({
  statementCount: 0,
  rows: [] as unknown[][],
}))

const proxyDb = drizzle(() => {
  state.statementCount += 1
  return Promise.resolve({ rows: state.rows })
})

vi.mock('../db/context.js', () => ({
  withUserContext: <T>(_userId: string, fn: (tx: UserTx) => Promise<T>): Promise<T> =>
    fn(proxyDb as unknown as UserTx),
}))

const { pushTokensDal } = await import('./push-tokens.js')

const USER_ID = '9b2b1c7e-2a44-4a3e-8f5d-6c1a2b3c4d5e'
const COLUMNS = Object.keys(getTableColumns(pushDeviceTokens))

// Deterministic uuid-shaped ids: 32 hex digits derived from the row index.
const rowId = (i: number) => {
  const hex = i.toString(16).padStart(12, '0')
  return `00000000-0000-4000-8000-${hex}`
}

const driverRows = (count: number): unknown[][] =>
  Array.from({ length: count }, (_, i) => {
    const row: Record<string, unknown> = {
      createdAt: `2026-01-01 00:00:${String(59 - (i % 60)).padStart(2, '0')}.${String(
        999999 - i,
      ).padStart(6, '0')}+00`,
      id: rowId(i),
      ownerId: USER_ID,
      platform: i % 2 === 0 ? 'android' : 'ios',
      token: `ExponentPushToken[device-${String(i)}]`,
      updatedAt: '2026-01-01 00:00:00.000000+00',
    }
    return COLUMNS.map((column) => row[column])
  })

describe('statement-count invariance (N+1 gate)', () => {
  it('register() is exactly ONE statement — the upsert needs no read-before-write', async () => {
    state.rows = driverRows(1)
    state.statementCount = 0
    const row = await pushTokensDal.register(USER_ID, {
      platform: 'android',
      token: 'ExponentPushToken[device-0]',
    })
    // Non-vacuous: the parsed row really flowed back out of the upsert.
    expect(row.ownerId).toBe(USER_ID)
    expect(state.statementCount).toBe(1)
  })

  it('list() is exactly ONE statement when the page comes back empty', async () => {
    state.rows = []
    state.statementCount = 0
    const page = await pushTokensDal.list(USER_ID, { limit: 20 })
    expect(page).toEqual({ items: [], nextCursor: null })
    expect(state.statementCount).toBe(1)
  })

  it('list() is STILL exactly ONE statement for a full page + sentinel (101 rows)', async () => {
    state.rows = driverRows(101)
    state.statementCount = 0
    const page = await pushTokensDal.list(USER_ID, { limit: 100 })
    // Non-vacuous: full page, has-more cursor — the sentinel row was consumed
    // in process, never via a second query.
    expect(page.items).toHaveLength(100)
    expect(page.nextCursor).not.toBeNull()
    expect(state.statementCount).toBe(1)
  })

  it('list() with a cursor stays ONE statement (the seek is part of the same SELECT)', async () => {
    state.rows = driverRows(3)
    state.statementCount = 0
    const page = await pushTokensDal.list(USER_ID, {
      limit: 20,
      cursor: { createdAt: '2026-01-01 00:00:30.000000+00', id: rowId(1) },
    })
    expect(page.items).toHaveLength(3)
    expect(state.statementCount).toBe(1)
  })

  it('remove() is exactly ONE statement, and reports the miss without a probe query', async () => {
    state.rows = []
    state.statementCount = 0
    const removed = await pushTokensDal.remove(USER_ID, rowId(7))
    expect(removed).toBe(false)
    expect(state.statementCount).toBe(1)
  })
})
